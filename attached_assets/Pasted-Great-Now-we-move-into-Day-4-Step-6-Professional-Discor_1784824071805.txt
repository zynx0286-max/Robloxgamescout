```python
import discord


def create_game_embed(game):

    embed = discord.Embed(
        title=f"🎮 {game['name']}",
        description="Roblox Game Opportunity",
    )

    embed.add_field(
        name="👥 Players",
        value=f"{game['playing']:,}",
        inline=True
    )

    embed.add_field(
        name="👀 Visits",
        value=f"{game['visits']:,}",
        inline=True
    )

    embed.add_field(
        name="⭐ Score",
        value=f"{game.get('score', 0)}/100",
        inline=True
    )


    embed.add_field(
        name="📈 Growth",
        value=f"+{game.get('growth',0)}%",
        inline=True
    )


    embed.add_field(
        name="🔥 Trend Score",
        value=f"{game.get('trend_score',0)}/100",
        inline=True
    )


    return embed
```

# Step 6.2 — Create Buttons

Create:

```text
buttons.py
```

Paste:

```python
import discord


class GameButtons(discord.ui.View):

    def __init__(self):
        super().__init__()


    @discord.ui.button(
        label="💾 Save Game",
        style=discord.ButtonStyle.green
    )
    async def save(
        self,
        interaction: discord.Interaction,
        button: discord.ui.Button
    ):

        await interaction.response.send_message(
            "Game saved!",
            ephemeral=True
        )


    @discord.ui.button(
        label="🔗 View Roblox",
        style=discord.ButtonStyle.link,
        url="https://www.roblox.com"
    )
    async def view_game(
        self,
        interaction,
        button
    ):
        pass
```
