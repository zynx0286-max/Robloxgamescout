
```text

utils.py
```

Paste:

```python
def calculate_score(game):
    score = 0

    # Player activity
    if game["playing"] >= 1000:
        score += 25
    elif game["playing"] >= 100:
        score += 10

    # Visits
    if game["visits"] >= 1000000:
        score += 25
    elif game["visits"] >= 100000:
        score += 10

    # Favorites
    if game["favorites"] >= 50000:
        score += 20
    elif game["favorites"] >= 10000:
        score += 10

    return score


def rank_games(games):

    for game in games:
        game["score"] = calculate_score(game)

    games.sort(
        key=lambda x: x["score"],
        reverse=True
    )

    return games
```

# Step 10 — Add `/scoretest` Command

This lets you test the scoring inside Discord.

Add:

```python
@bot.tree.command(
    name="scoretest",
    description="Test the game scoring system"
)
async def scoretest(interaction: discord.Interaction):

    from scanner import calculate_score

    game = {
        "playing": 5000,
        "visits": 5000000,
        "favorites": 100000
    }

    score = calculate_score(game)

    await interaction.response.send_message(
        f"Game Scout Score: {score}/100"
    )
```
